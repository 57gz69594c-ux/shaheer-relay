// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

contract AaveFlashLiquidator {
    IPool public immutable POOL;
    ISwapRouter public immutable SWAP_ROUTER;
    address public owner;
    mapping(address => bool) public keepers;

    uint8 constant LIQ_AAVE = 0;
    uint8 constant LIQ_COMPOUND = 1;

    struct LiqParams {
        uint8 liqType;
        address collateralAsset;
        address debtAsset;
        address borrower;
        uint256 debtToCover;
        address cTokenDebt;
        address cTokenCollateral;
        uint24 swapFee;
        uint256 minSwapOut;
    }

    constructor(address _pool, address _swapRouter) {
        POOL = IPool(_pool);
        SWAP_ROUTER = ISwapRouter(_swapRouter);
        owner = msg.sender;
        keepers[msg.sender] = true;
    }

    function liquidate(LiqParams calldata params) external onlyKeeper {
        POOL.flashLoanSimple(
            address(this),
            params.debtAsset,
            params.debtToCover,
            abi.encode(params),
            0
        );
    }

    function executeOperation(
        address asset, uint256 amount, uint256 premium,
        address, bytes calldata params
    ) external returns (bool) {
        require(msg.sender == address(POOL), "NOT_POOL");
        LiqParams memory p = abi.decode(params, (LiqParams));

        uint256 collateralBefore = IERC20(p.collateralAsset).balanceOf(address(this));

        if (p.liqType == LIQ_AAVE) {
            IERC20(p.debtAsset).approve(address(POOL), amount);
            POOL.liquidationCall(p.collateralAsset, p.debtAsset, p.borrower, p.debtToCover, false);
        } else if (p.liqType == LIQ_COMPOUND) {
            IERC20(p.debtAsset).approve(p.cTokenDebt, amount);
            uint256 err = ICToken(p.cTokenDebt).liquidateBorrow(p.borrower, p.debtToCover, p.cTokenCollateral);
            require(err == 0, "COMPOUND_LIQ_FAILED");
            uint256 cTokenBalance = ICToken(p.cTokenCollateral).balanceOf(address(this));
            require(cTokenBalance > 0, "NO_CTOKENS_SEIZED");
            uint256 redeemErr = ICToken(p.cTokenCollateral).redeem(cTokenBalance);
            require(redeemErr == 0, "COMPOUND_REDEEM_FAILED");
        }

        uint256 collateralSeized = IERC20(p.collateralAsset).balanceOf(address(this)) - collateralBefore;
        require(collateralSeized > 0, "NO_COLLATERAL_SEIZED");

        // Swap collateral -> debt asset via Uniswap V3
        IERC20(p.collateralAsset).approve(address(SWAP_ROUTER), collateralSeized);
        SWAP_ROUTER.exactInputSingle(ISwapRouter.ExactInputSingleParams({
            tokenIn: p.collateralAsset,
            tokenOut: asset,
            fee: p.swapFee,
            recipient: address(this),
            amountIn: collateralSeized,
            amountOutMinimum: p.minSwapOut,
            sqrtPriceLimitX96: 0
        }));

        uint256 owed = amount + premium;
        require(IERC20(asset).balanceOf(address(this)) >= owed, "UNPROFITABLE");
        IERC20(asset).approve(address(POOL), owed);
        return true;
    }

    // CRITICAL ISSUE: The contract takes flash loan from the SAME Aave pool
    // it's trying to liquidate on. The POOL address passed to constructor
    // is used for BOTH flashLoanSimple() AND liquidationCall().
    // This works IF the pool supports both. But the pool_data_provider
    // addresses in fleet_config.py may be wrong — see BUG A above.
}

contract FlashLiquidator {
    IMorpho public immutable morpho;
    ISwapRouter02 public immutable router;
    address public owner;
    mapping(address => bool) public keepers;
    uint256 public maxSlippageWad = 0.08e18;

    struct LiquidateParams {
        MarketParams marketParams;
        address borrower;
        uint256 seizedAssets;
        uint256 repaidShares;
        uint24 swapFee;
        uint256 flashAmount;
        uint256 minAmountOut;
    }

    function liquidate(LiquidateParams calldata params) external onlyOwnerOrKeeper {
        morpho.flashLoan(params.marketParams.loanToken, params.flashAmount, abi.encode(params));
    }

    function onMorphoFlashLoan(uint256 assets, bytes calldata data) external onlyMorpho {
        LiquidateParams memory params = abi.decode(data, (LiquidateParams));
        IERC20(params.marketParams.loanToken).approve(address(morpho), params.flashAmount);
        (uint256 seized, uint256 repaid) = morpho.liquidate(
            params.marketParams, params.borrower,
            params.seizedAssets, params.repaidShares, ""
        );
        // Swap collateral -> loan token
        uint256 collateralBalance = IERC20(params.marketParams.collateralToken).balanceOf(address(this));
        IERC20(params.marketParams.collateralToken).approve(address(router), collateralBalance);
        router.exactInputSingle(ISwapRouter02.ExactInputSingleParams({
            tokenIn: params.marketParams.collateralToken,
            tokenOut: params.marketParams.loanToken,
            fee: params.swapFee,
            recipient: address(this),
            amountIn: collateralBalance,
            amountOutMinimum: params.minAmountOut,
            sqrtPriceLimitX96: 0
        }));
        require(IERC20(params.marketParams.loanToken).balanceOf(address(this)) >= assets, "UNPROFITABLE");
        IERC20(params.marketParams.loanToken).approve(address(morpho), assets);
    }
}
